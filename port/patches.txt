# Exact-text substitutions applied to the port's source mirror by
# port/tools/mirror_src.py.  The decomp's own files are never edited.
#
#   [target:]relative/path<TAB>old text<TAB>new text
#
# `\n` and `\t` are unescaped in both texts.  Each patch must match its file
# exactly once or the build fails, so an upstream edit that moves the text is
# a loud error rather than a silent no-op.  `host:` / `ppc:` restricts a patch
# to one target.
#
# Metrowerks `asm { }` blocks and whole-function `asm` definitions are removed
# by the mirror itself, not from here -- see mirror_src.py step 2.

# ---------------------------------------------------------------- game/malloc.c
# Four `mflr r3` blocks tag each allocation with its caller's return address.
# The mirror strips the asm; these restore the value the portable way.  (The
# tag is only ever printed by HuMemHeapDump, so truncating a 64-bit host
# address to the u32 field is harmless.)
x4:src/game/malloc.c	    register u32 retaddr;\n    asm {\n        mflr retaddr\n    }\n	    u32 retaddr = (u32)(uintptr_t)__builtin_return_address(0);\n

# --------------------------------------------------------------- game/process.c
# HUPROCESS coroutine stacks.  The default is 2,048 bytes and `base_sp` leaves
# only 8 bytes of headroom above the top of the allocation, which is exactly
# enough for the Metrowerks EABI prologue and not enough for a Darwin one
# (PowerPC Darwin stores LR at 8(r1) of the caller's frame; arm64 wants 16-byte
# alignment).  Multiply the stacks -- a GCC 14 frame is fatter than a
# Metrowerks EABI frame -- and give the fake frame real headroom.  The game's
# own "stack overlap error" guard byte in HuPrcCall still checks the result.
src/game/process.c	    if(stack_size == 0) {\n        stack_size = 2048;\n    }\n	    if(stack_size == 0) {\n        stack_size = 2048;\n    }\n    stack_size *= PORT_PRC_STACK_MUL;\n

# ---------------------------------------------------------------- game/objdll.c
# The REL loader's failure path falls through and calls the module's prolog
# anyway, at whatever `module->prolog` happened to contain -- which, with no
# relocation done, is a wild jump.  M1's OSLink deliberately fails (the real
# loader is one dlopen'ed Mach-O bundle per REL, which is M2), so the failure
# has to be survivable: report it, dump the module info, and return without
# entering the module.  This is also exactly the seam M2 replaces.
src/game/objdll.c	\tif(OSLink(&dll->module->info, dll->bss) != TRUE) {\n\t\tOSReport("objdll>++++++++++++++++ DLL Link Failed\\n");\n\t}\n\tomDLLInfoDump(&dll->module->info);\n	\tif(OSLink(&dll->module->info, dll->bss) != TRUE) {\n\t\tOSReport("objdll>++++++++++++++++ DLL Link Failed\\n");\n\t\tomDLLInfoDump(&dll->module->info);\n\t\tomDLLHeaderDump(dll->module);\n\t\tdll->ret = 0;\n\t\treturn dll;\n\t}\n\tomDLLInfoDump(&dll->module->info);\n
