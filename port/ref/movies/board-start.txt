# board-start -- boot to the first minigame of Toad's Midway Madness.
#
# Compile:  ../tools/mkgecko.py board-start.txt "$MP4_USERDIR/GameSettings/GMPE01.ini" --name RefBoardStart
# Capture:  ../tools/capture.sh 700 /tmp/board -C Dolphin.Core.EnableCheats=True
#
# Frame numbers are the game's own GlobalCounter (0x801D3A54), which is also
# what the port's --play schedule counts, so this and
# ../movies/board-start.play are the same walk with the same shape -- START
# through the boot, then A for 4 frames out of every 64, forever.  The two
# sides do not agree on absolute frames (the port skips the DVD seek and the
# opening movie), so only the shape can be shared, and the metronome is what
# makes that possible: no prompt in the run has to be found.
#
# The metronome is one `every` rather than 200 `at`s on purpose -- see
# mkgecko.py.  A list that overflows Dolphin's Gecko region is silently not
# installed, and the capture then shows the attract loop with no error
# anywhere.  That cost a run.
#
# Requires a freshly created memory card in slot A: the game will not pass
# SELECT A FILE without one.

# Hold START through the logos and the 70 s opening movie.
at 10 320 START

# Accept the title (three pulses: the frame the title starts taking input is
# not worth pinning down).
at 380 6 START
at 430 6 START
at 480 6 START

# A, four frames out of every sixty-four, from the file select onwards: the
# new-file scene, the Party Mode prompts, character select, the board
# settings, the board intro, the turn-order roll, the first turn, and the
# first minigame.
every 64 4 560 A

# The minigame instruction screen ends on START and nothing else
# (src/REL/instDll/main.c:293 tests `btnDown == PAD_BUTTON_START`; the
# auto-start beside it only fires when all four players are CPU, and this walk
# leaves one human).  Getting that one press right cost four captures:
#
#   * it cannot share a frame with an A -- instDll compares `btnDown` for
#     equality against PAD_BUTTON_START, and A+START is 0x1100.  Hence +32.
#   * it cannot be a metronome over the whole run.  A START anywhere else opens
#     the *board* pause menu (src/game/board/pause.c:1493), which sleeps before
#     it reads input, so the next A walks into "Please choose which character's
#     settings to change" and the walk stays there to the end.  Both the port
#     and Dolphin got stuck there, identically, which is its own small piece of
#     evidence about the port.
#   * it cannot be one or two taps at a pinned frame either: the screen does
#     not take input until its entry animation finishes (`while (instMode != 1)`),
#     which is a few hundred frames after it first appears.
#
# So it is a metronome inside a window -- from after the first minigame starts
# to before it can possibly be over -- which is the one part of the walk pinned
# to frame numbers, and pinned to an observation from an A-only capture rather
# than a guess.
every 256 1 12000 START 32 14000
